## 一、新增表结构

### 1.1 标签定义表 `tag_definition`

解决问题：标签类型（秘境、珍馐等）不可配置的问题

```sql
CREATE TABLE `tag_definition` (
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT,
  `tag_code` VARCHAR(50) NOT NULL COMMENT '标签代码（唯一）',
  `tag_name` VARCHAR(100) NOT NULL COMMENT '标签显示名称',
  `tag_icon` VARCHAR(255) DEFAULT NULL COMMENT '标签图标URL',
  `tag_color` VARCHAR(20) DEFAULT NULL COMMENT '标签颜色（如 #FF5722）',
  `tag_emoji` VARCHAR(10) DEFAULT NULL COMMENT '标签emoji',
  `min_level_code` VARCHAR(50) NOT NULL DEFAULT 'LANDONG' COMMENT '最低可贴等级代码',
  `description` VARCHAR(500) DEFAULT NULL COMMENT '标签描述',
  `sort_order` INT(11) DEFAULT 0 COMMENT '排序顺序',
  `is_active` TINYINT(1) DEFAULT 1 COMMENT '是否启用',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_tag_code` (`tag_code`),
  KEY `idx_active_sort` (`is_active`, `sort_order`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='标签定义表';
```

### 1.2 奖章触发条件表 `medal_condition`

解决问题：奖章条件硬编码的问题

```sql
CREATE TABLE `medal_condition` (
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT,
  `medal_code` VARCHAR(50) NOT NULL COMMENT '关联的奖章代码',
  `condition_type` VARCHAR(50) NOT NULL COMMENT '条件类型',
  `condition_params` JSON NOT NULL COMMENT 'JSON格式的条件参数',
  `description` VARCHAR(255) DEFAULT NULL COMMENT '条件描述',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_medal_code` (`medal_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='奖章触发条件表';

-- 条件类型枚举：
-- VIEW_SHARE_TOTAL: 累计查看分享数
-- VIEW_SHARE_CONSECUTIVE: 连续查看分享数（需配合时长）
-- CHECKIN_TOTAL: 累计打卡数
-- CHECKIN_TAG: 打卡特定标签的分享数
-- COMMENT_TOTAL: 累计评论数
-- POST_SHARE_TOTAL: 累计发布分享数
```

### 1.3 等级复合条件表 `level_condition`

解决问题：等级条件（如"打卡某一标签的分享>100"）的配置

```sql
-- 扩展 level_definition 表或新建 level_condition 表
-- 方案：扩展 level_definition 表增加 JSON 字段

ALTER TABLE `level_definition`
ADD COLUMN `extra_conditions` JSON DEFAULT NULL COMMENT 'JSON格式的额外条件';

-- 更新水母等级的额外条件
UPDATE `level_definition`
SET `extra_conditions` = '{"checkin_any_tag_min": 100}'
WHERE `level_code` = 'SHUIMU';
```

### 1.4 分享举报表 `share_report`

解决问题：举报流程管理

```sql
CREATE TABLE `share_report` (
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT,
  `share_id` BIGINT(20) NOT NULL COMMENT '被举报的分享ID',
  `reporter_id` BIGINT(20) NOT NULL COMMENT '举报人ID',
  `reason_type` VARCHAR(50) NOT NULL COMMENT '举报原因类型',
  `reason_detail` TEXT DEFAULT NULL COMMENT '举报详情',
  `status` TINYINT(4) NOT NULL DEFAULT 0 COMMENT '处理状态: 0=待处理, 1=已确认违规, 2=已驳回',
  `handler_id` BIGINT(20) DEFAULT NULL COMMENT '处理人ID',
  `handle_result` VARCHAR(255) DEFAULT NULL COMMENT '处理结果',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `handled_at` DATETIME DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_share_id` (`share_id`),
  KEY `idx_status` (`status`),
  KEY `idx_reporter` (`reporter_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='分享举报表';
```

### 1.5 管理操作日志表 `admin_operation_log`

解决问题：管理操作审计

```sql
CREATE TABLE `admin_operation_log` (
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT,
  `admin_id` BIGINT(20) NOT NULL COMMENT '操作管理员ID',
  `operation_type` VARCHAR(50) NOT NULL COMMENT '操作类型',
  `target_type` VARCHAR(50) NOT NULL COMMENT '目标类型（USER/SHARE/CONFIG）',
  `target_id` BIGINT(20) DEFAULT NULL COMMENT '目标ID',
  `before_value` JSON DEFAULT NULL COMMENT '操作前的值',
  `after_value` JSON DEFAULT NULL COMMENT '操作后的值',
  `remark` VARCHAR(255) DEFAULT NULL COMMENT '操作备注',
  `ip_address` VARCHAR(50) DEFAULT NULL COMMENT '操作IP',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_admin_id` (`admin_id`),
  KEY `idx_target` (`target_type`, `target_id`),
  KEY `idx_created_at` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='管理操作日志表';
```

### 2.2 扩展 userlist 表

```sql
-- 新增用户统计字段（用于等级计算）
ALTER TABLE `userlist`
ADD COLUMN `checkin_count_total` INT(11) DEFAULT 0 COMMENT '累计打卡次数',
ADD COLUMN `comment_count_total` INT(11) DEFAULT 0 COMMENT '累计评论次数',
ADD COLUMN `view_count_total` INT(11) DEFAULT 0 COMMENT '累计查看分享次数',
ADD COLUMN `tag_count_total` INT(11) DEFAULT 0 COMMENT '累计贴标签次数';
```

```sql
-- 新增查看时长字段（用于奖章判断）
```
